using System;
using System.ComponentModel;
using System.Windows;
using System.Windows.Controls;
using System.Windows.Input;
using System.Windows.Media;
using System.Windows.Media.Imaging;
using System.Windows.Threading;

namespace GrayMatch.Wpf;

/// <summary>
/// 图像查看器的缩放与平移（滚轮缩放 / 中键拖动 / 双击复位）。
///
/// 实现要点：不用 <c>ScrollViewer</c> 的滚动偏移量做平移。
/// <c>ScrollViewer</c> 在“内容比视口小”时 ExtentWidth == ViewportWidth、Offset 恒为 0，
/// 滚动请求会被钳死，中键拖动看起来毫无反应；缩放时内容还会被重新居中，光标锚点漂移。
///
/// 现在改为纯渲染变换：<c>ImageGrid.RenderTransform = TransformGroup(Scale × Translate)</c>，
/// 缩放/平移全部自己算，坐标系完全可控：
///   - 平移只改 TranslateX/Y，与布局无关，不会被任何滚动逻辑钳位；
///   - 缩放围绕光标：content = cursor − T；T' = cursor − content × S'；
///   - 复位（双击 / 打开新图）＝ 窗口最大化 + 按视口自适应并居中。
///
/// 坐标系：ImageGrid 的逻辑尺寸 == 图像显示尺寸，overlay（ROI/结果/缺陷）都活在图像坐标系里，
/// <c>Mouse.GetPosition(ImageGrid)</c> 返回未缩放的图像坐标，因此 ROI/结果框始终贴合像素。
///
/// 操作：
///   - 滚轮        → 以光标为中心缩放
///   - 中键拖动    → 平移
///   - 左键拖动    → 框选 ROI 模板（MainWindow.xaml.cs 处理）
///   - 双击左键    → 复位：最大化 + 自适应铺满居中
///   - 打开新图 / 缩放模式下改变窗口大小 → 自动适配
/// </summary>
public partial class MainWindow
{
    private const double ZoomMin = 0.02;         // 最小缩放（相对图像原始尺寸）
    private const double ZoomMax = 64.0;         // 最大缩放
    private const double ZoomStep = 1.15;        // 每一格滚轮的缩放步进
    private const double PanKeepVisible = 80.0;  // 平移时至少保留在视口内的像素数

    private bool _isPanning;
    private bool _panMoved;           // 本次中键拖动是否真的移动过
    private Point _panStart;          // 上一次鼠标位置（视口坐标）
    private bool _manualTransform;    // 用户已手动缩放/平移 → 不再自动适配
    private bool _initialFitPending = true;
    private bool _fitQueued;          // RequestFit 去重
    private bool _isFitting;          // FitToView 重入保护
    private bool _loggedFit;          // 首帧适配写一次状态栏
    private double _contentW;         // 当前图像显示宽度（设备无关单位）
    private double _contentH;

    private void ImageViewer_Loaded(object sender, RoutedEventArgs e)
    {
        // LayoutUpdated 是“布局已完成”的可靠信号（每次 arrange 后触发，用标志控制只适配一次）。
        ImageGrid.LayoutUpdated += ImageGrid_LayoutUpdated;

        // 换图时重新适配。
        DependencyPropertyDescriptor.FromProperty(Image.SourceProperty, typeof(Image))
            .AddValueChanged(SourceImage, OnSourceImageChanged);

        // 视口尺寸变化后重新适配。
        ImageViewport.SizeChanged += ImageViewport_SizeChanged;

        // 首帧兜底：窗口刚显示时 ActualWidth 可能还是 0，等布局稳下来再适配一次。
        Dispatcher.BeginInvoke(DispatcherPriority.ApplicationIdle, new Action(RequestFit));

        // 窗口从设计尺寸（1320×840）切到最大化尺寸属于“迟到的布局”，也必须重新居中。
        ScheduleRefit(4);
    }

    private void ImageViewport_SizeChanged(object sender, SizeChangedEventArgs e)
    {
        // 只有用户还没手动缩放/平移过才跟着适配；手动调整后窗口大小变化不动视图。
        if (_manualTransform || _isPanning) return;
        RequestFit();
    }

    /// <summary>安排一次“适配视口”，同一拍内多次请求只执行一次。</summary>
    private void RequestFit()
    {
        if (_fitQueued) return;
        _fitQueued = true;
        Dispatcher.BeginInvoke(DispatcherPriority.Loaded, new Action(() =>
        {
            _fitQueued = false;
            if (!_manualTransform && !_isPanning) FitToView();
        }));
    }

    /// <summary>
    /// 延迟若干帧后再量一次视口/图像尺寸重新居中。WindowState=Maximized 的窗口启动时
    /// 往往先按设计尺寸（1320×840）布局一拍再切到最大化尺寸；不补这几拍，
    /// 居中偏移就是按小视口算的，窗口变大后图像就偏到右下角去。
    /// </summary>
    private void ScheduleRefit(int frames)
    {
        Dispatcher.BeginInvoke(DispatcherPriority.Loaded, new Action(() =>
        {
            if (_manualTransform || _isPanning) return;
            FitToView();
            if (frames > 1) ScheduleRefit(frames - 1);
        }));
    }

    private void ImageGrid_LayoutUpdated(object? sender, EventArgs e)
    {
        // 首帧兜底：还没适配过且视口已量出尺寸，就立刻适配。
        if (_initialFitPending && !_manualTransform)
            FitToView();
    }

    private void OnSourceImageChanged(object? sender, EventArgs e)
    {
        // 新图一律“最大化 + 适配视口”，无论之前缩放/平移到什么状态。
        _manualTransform = false;
        _initialFitPending = true;
        _isPanning = false;

        EnsureMaximized();

        FitToView();
        RequestFit();
        ScheduleRefit(6);
    }

    private void ImageGrid_SizeChanged(object sender, SizeChangedEventArgs e)
    {
        if (_isFitting || _manualTransform) return;
        RequestFit();
    }

    /// <summary>让窗口处于最大化状态（“自适应最大化”）；已经是最大化就不动，避免打扰用户。</summary>
    private void EnsureMaximized()
    {
        if (WindowState != WindowState.Maximized)
            WindowState = WindowState.Maximized;
    }

    /// <summary>
    /// 复位：最大化窗口 + 自适应铺满并居中。视口尺寸要等最大化布局完成才准，
    /// 所以先立即适配一次（响应即时），再补几拍（最终严格铺满居中）。
    /// </summary>
    private void ResetView()
    {
        _manualTransform = false;
        _initialFitPending = true;
        _loggedFit = false;
        EnsureMaximized();
        FitToView();
        RequestFit();
        ScheduleRefit(6);
    }

    // ---------------------------------------------------------------- 滚轮缩放

    private void ImageArea_MouseWheel(object sender, MouseWheelEventArgs e)
    {
        if (SourceImage?.Source == null) return;

        double oldScale = ImageScaleTransform.ScaleX;
        if (oldScale <= 0) oldScale = 1.0;

        double factor = e.Delta > 0 ? ZoomStep : 1.0 / ZoomStep;
        double newScale = Math.Max(ZoomMin, Math.Min(ZoomMax, oldScale * factor));
        if (Math.Abs(newScale - oldScale) < 1e-9)
        {
            e.Handled = true;
            return;
        }

        // 光标位置，以及它对应的“内容（图像）坐标”。
        Point cursor = e.GetPosition(ImageViewport);
        double contentX = (cursor.X - ImageTranslateTransform.X) / oldScale;
        double contentY = (cursor.Y - ImageTranslateTransform.Y) / oldScale;

        // 缩放后让同一个内容点仍然停在光标下。
        ImageTranslateTransform.X = cursor.X - contentX * newScale;
        ImageTranslateTransform.Y = cursor.Y - contentY * newScale;

        ImageScaleTransform.ScaleX = newScale;
        ImageScaleTransform.ScaleY = newScale;

        ClampTranslate();

        _manualTransform = true;
        StatusText = $"缩放 {newScale * 100:F0}%";
        e.Handled = true;
    }

    // ---------------------------------------------------------------- 中键拖动

    private void ImageArea_MouseDown(object sender, MouseButtonEventArgs e)
    {
        // 左键双击 → 复位：最大化 + 自适应铺满居中。
        if (e.ChangedButton == MouseButton.Left && e.ClickCount == 2)
        {
            ResetView();
            e.Handled = true;
            return;
        }

        // 中键按下 → 开始平移。捕获到视口元素本身，拖到窗口外也不丢事件。
        // 注意：这里不置 _manualTransform，只有真的拖动了才算“用户手动调整过”，
        // 否则轻轻点一下中键就会把后续自动适配关掉。
        if (e.ChangedButton == MouseButton.Middle)
        {
            _isPanning = true;
            _panMoved = false;
            _panStart = e.GetPosition(ImageViewport);
            ImageViewport.CaptureMouse();
            ImageViewport.Cursor = Cursors.SizeAll;
            e.Handled = true;
        }
    }

    private void ImageArea_MouseMove(object sender, MouseEventArgs e)
    {
        if (!_isPanning) return;

        // 捕获丢失等极端情况下中键已抬起 → 收尾，避免状态卡住。
        if (e.MiddleButton == MouseButtonState.Released)
        {
            EndPan();
            return;
        }

        Point p = e.GetPosition(ImageViewport);
        if (!_panMoved && Math.Abs(p.X - _panStart.X) < 0.5 && Math.Abs(p.Y - _panStart.Y) < 0.5)
            return;   // 还没真正拖动，保持自动适配状态

        _panMoved = true;
        _manualTransform = true;   // 真正拖动过 → 停止自动适配
        ImageTranslateTransform.X += p.X - _panStart.X;
        ImageTranslateTransform.Y += p.Y - _panStart.Y;
        _panStart = p;
        ClampTranslate();
        e.Handled = true;
    }

    private void ImageArea_MouseUp(object sender, MouseButtonEventArgs e)
    {
        if (!_isPanning) return;

        // 中键抬起，或按键状态已全部松开（事件丢失的兜底）→ 结束平移。
        if (e.ChangedButton == MouseButton.Middle ||
            (e.MiddleButton == MouseButtonState.Released && e.LeftButton == MouseButtonState.Released))
        {
            EndPan();
            e.Handled = true;
        }
    }

    private void ImageArea_LostMouseCapture(object sender, MouseEventArgs e)
    {
        // Alt+Tab / 弹窗等导致捕获丢失时收尾，避免状态卡在“平移中”。
        if (_isPanning) EndPan();
    }

    private void EndPan()
    {
        _isPanning = false;
        _panMoved = false;
        if (ImageViewport.IsMouseCaptured) ImageViewport.ReleaseMouseCapture();
        ImageViewport.Cursor = Cursors.Arrow;
    }

    // ---------------------------------------------------------------- 适配视口

    /// <summary>
    /// 按“整幅图都能看见”的比例缩放并居中。双击复位、打开新图、视口尺寸变化都用它。
    ///
    /// 尺寸用 <see cref="ImageSource.Width"/> 再按 (96 / DPI) 换算成设备无关单位：
    /// 图像带 96 以外 DPI 时 WPF 会按 DPI 缩放显示，直接按像素数算会把内容尺寸估大、
    /// 居中偏移算歪，图像就跑到右下角去。
    /// </summary>
    private void FitToView()
    {
        if (_isFitting) return;
        _isFitting = true;
        try
        {
            var src = SourceImage?.Source;
            if (src == null) return;

            double contentW = ContentWidth(src);
            double contentH = ContentHeight(src);
            if (contentW <= 0.5 || contentH <= 0.5) return;

            double vpW = ImageViewport.ActualWidth;
            double vpH = ImageViewport.ActualHeight;

            // 视口还没量出尺寸（启动首帧）→ 保持待适配，等布局完成后再来一次。
            if (vpW <= 1 || vpH <= 1)
            {
                _initialFitPending = true;
                return;
            }

            _contentW = contentW;
            _contentH = contentH;
            _initialFitPending = false;

            // 网格尺寸 = 图像显示尺寸。Grid 没有显式尺寸时会拉伸填满视口，
            // 大图就会被“视口大小的容器”裁掉一大半，看上去像跑到右下角。
            // 注意 Width 初始是 NaN，NaN 参与比较恒为 false，必须单独判 NaN。
            if (double.IsNaN(ImageGrid.Width) || Math.Abs(ImageGrid.Width - contentW) > 0.01)
                ImageGrid.Width = contentW;
            if (double.IsNaN(ImageGrid.Height) || Math.Abs(ImageGrid.Height - contentH) > 0.01)
                ImageGrid.Height = contentH;

            // 让 <Image> 自身也按同一尺寸渲染：位图 DPI 元数据与容器尺寸不一致时，
            // 位图和容器会错位（另一个“偏右下角”的来源）。
            if (SourceImage != null)
            {
                if (double.IsNaN(SourceImage.Width) || Math.Abs(SourceImage.Width - contentW) > 0.01)
                    SourceImage.Width = contentW;
                if (double.IsNaN(SourceImage.Height) || Math.Abs(SourceImage.Height - contentH) > 0.01)
                    SourceImage.Height = contentH;
            }

            double scale = Math.Min(vpW / contentW, vpH / contentH);
            if (scale <= 0 || double.IsInfinity(scale)) return;
            scale = Math.Max(ZoomMin, Math.Min(ZoomMax, scale));

            ImageScaleTransform.ScaleX = scale;
            ImageScaleTransform.ScaleY = scale;

            // 居中：内容中心对齐视口中心（左右、上下都居中）。
            ImageTranslateTransform.X = (vpW - contentW * scale) / 2.0;
            ImageTranslateTransform.Y = (vpH - contentH * scale) / 2.0;

            if (!_loggedFit)
            {
                _loggedFit = true;
                StatusText = $"视图复位：缩放={scale:F3} 图像={contentW:F0}x{contentH:F0} 视口={vpW:F0}x{vpH:F0}";
            }
        }
        finally
        {
            _isFitting = false;
        }
    }

    /// <summary>图像显示宽度（设备无关单位 = 像素 × 96 / DPI）。</summary>
    private static double ContentWidth(ImageSource src)
    {
        var bmp = src as BitmapSource;
        double dpi = (bmp != null && bmp.DpiX > 1) ? bmp.DpiX : 96.0;
        return src.Width * 96.0 / dpi;
    }

    /// <summary>图像显示高度（设备无关单位 = 像素 × 96 / DPI）。</summary>
    private static double ContentHeight(ImageSource src)
    {
        var bmp = src as BitmapSource;
        double dpi = (bmp != null && bmp.DpiY > 1) ? bmp.DpiY : 96.0;
        return src.Height * 96.0 / dpi;
    }

    /// <summary>
    /// 平移限位：允许把图拖到视口外，但至少留 <see cref="PanKeepVisible"/> 像素可见，
    /// 免得“一拖就再也找不回来”。内容比视口小时允许完整居中偏移。
    /// </summary>
    private void ClampTranslate()
    {
        var src = SourceImage?.Source;
        if (src == null) return;

        double scale = ImageScaleTransform.ScaleX;
        if (scale <= 0) return;

        double vpW = ImageViewport.ActualWidth;
        double vpH = ImageViewport.ActualHeight;
        if (vpW <= 1 || vpH <= 1) return;

        double contentW = (_contentW > 0 ? _contentW : ContentWidth(src)) * scale;
        double contentH = (_contentH > 0 ? _contentH : ContentHeight(src)) * scale;

        ImageTranslateTransform.X = ClampOffset(ImageTranslateTransform.X, PanKeepVisible - contentW, vpW - PanKeepVisible);
        ImageTranslateTransform.Y = ClampOffset(ImageTranslateTransform.Y, PanKeepVisible - contentH, vpH - PanKeepVisible);
    }

    private static double ClampOffset(double value, double min, double max)
        => max < min ? value : Math.Max(min, Math.Min(max, value));
}
